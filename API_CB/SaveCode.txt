    return {
        barcodeRequestEmail: row.BarcodeEmail || process.env.DEFAULT_BARCODE_EMAIL,

        shipperAddress: {

            poBox: "",

            stateCode: "",
            stateName: "",
            preferredLanguage: "",

            schenkerAddressIdentifierValue: "",

            street2: row.ShipperStreet2 || "",

        deliveryAddress: {

            contactPhone: "",
            contactEmail: "",
            name1: row.DeliveryName,

            stateCode: "",

            preferredLanguage: "",
            schenkerAddressId: "",
            schenkerAddressIdentifierType: "DELIVERY",
            schenkerAddressIdentifierValue: "",
            street: row.DeliveryStreet,
            street2: row.DeliveryStreet2 || "",
            city: row.DeliveryCity,
            countryCode: row.DeliveryCountryCode || "FR",
            type: "DELIVERY"
        },

        incoterm: "EXW",
        incotermLocation: row.IncotermLocation || "",
        productCode: "43",
        productOption: "",
        measurementType: "METRIC",
        customsClearance: false,
        grossWeight: row.GrossWeight || 0,
        indoorDelivery: false,
        pickUpDateFrom: row.PickupDateFrom,
        pickUpDateTo: row.PickupDateTo,
        referenceNumber: row.BLNumber,
        neutralShipping: false,
        specialCargo: false,
        valueOfGoods: row.ValueOfGoods || 0,
        valueOfGoodsCurrency: "EUR",
        ecoNeutral: true,
        serviceType: "D2D",
        dgr: false,
        cargoDesc: row.CargoDesc || "Expedition",
        volume: row.Volume || 0,
        marksAndNumbers: "",
        packageType: "pk",
        pieces: row.Pieces || 1,
        stackable: true,
        express: false,
        foodRelated: false,
        heatedTransport: false,
        homeDelivery: false,
        measureUnit: "PIECES",
        ownPickup: false,
        pharmaceuticals: false
    };
}
